# decisions D20–D28 in docs/plan.md §1 (D22–D24, D26, D28 amended 2026-10-24).

include("${PROJECT_SOURCE_DIR}/cmake/Vulkan.cmake")

# Isolation guard (D3/D4, vulkan-plan Layout): the Vulkan tree reaches only the
# backend-neutral frontend, params headers and (from V2b) src/gpuhost; never
# CPU kernels, CPU host code or the CUDA tree.
file(GLOB_RECURSE NSS_VK_GUARDED CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*")
nss_forbid_cpu_includes("Vulkan backend" ${NSS_VK_GUARDED})
foreach(_nss_file IN LISTS NSS_VK_GUARDED)
  if(_nss_file MATCHES "\\.(hpp|cpp|h|comp|glsl)$")
    file(STRINGS "${_nss_file}" _nss_hits REGEX "#[ \t]*include[ \t]+[\"<]cuda/")
    if(_nss_hits)
      message(FATAL_ERROR "Vulkan backend file includes CUDA code: ${_nss_file}: ${_nss_hits}")
    endif()
  endif()
endforeach()

# Shader variants (D24). Every .comp under shaders/ is compiled by at least one
# entry here; configure fails on an unlisted or missing source, as the CUDA
# flag-class whitelist does (D18).
set(NSS_VK_SHADER_IDS "")
set(NSS_VK_SHADER_SOURCES "")
set(NSS_VK_SPIRV_SOURCES "")
nss_vk_add_shader(NSS_VK_SPIRV_SOURCES runtime_probe runtime/probe.comp)

file(GLOB_RECURSE _nss_vk_all_comp CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/shaders"
     "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.comp")
set(_nss_vk_unlisted ${_nss_vk_all_comp})
if(NSS_VK_SHADER_SOURCES)
  list(REMOVE_ITEM _nss_vk_unlisted ${NSS_VK_SHADER_SOURCES})
endif()
if(_nss_vk_unlisted)
  message(FATAL_ERROR "Vulkan shaders without a variant entry in src/vulkan/CMakeLists.txt: ${_nss_vk_unlisted}")
endif()

# Registry of the embedded variants (runtime/shaders.hpp).
set(_nss_vk_decls "")
set(_nss_vk_rows "")
foreach(id IN LISTS NSS_VK_SHADER_IDS)
  string(APPEND _nss_vk_decls "extern const std::uint32_t ${id}[];\nextern const std::size_t ${id}_words;\n")
  string(APPEND _nss_vk_rows "    {\"${id}\", {spirv::${id}, spirv::${id}_words}},\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/spirv/registry.cpp" CONTENT
"// Generated by src/vulkan/CMakeLists.txt; do not edit.
#include \"vulkan/runtime/shaders.hpp\"

#include <stdexcept>
#include <string>

namespace nss_vk {
namespace spirv {
@_nss_vk_decls@}  // namespace spirv

namespace {
struct Entry {
    const char* id;
    Spirv code;
};
const Entry kShaders[] = {
@_nss_vk_rows@};
}  // namespace

Spirv shader(std::string_view id) {
    for (const Entry& entry : kShaders) {
        if (id == entry.id) return entry.code;
    }
    throw std::logic_error(\"nss_vk: no embedded shader '\" + std::string(id) + \"'\");
}

const char* shader_tools() {
    return \"@NSS_VK_SHADER_TOOLS@\";
}

}  // namespace nss_vk
" @ONLY)

add_library(nss_vk_shaders STATIC ${NSS_VK_SPIRV_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/spirv/registry.cpp")
target_include_directories(nss_vk_shaders PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
set_target_properties(nss_vk_shaders PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
)

# nss_vk_core: host runtime + embedded shaders, static so tests/vk can link it.
add_library(nss_vk_core STATIC
  runtime/device.cpp
  runtime/error.cpp
  runtime/instance.cpp
  runtime/probe.cpp
)
target_include_directories(nss_vk_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(nss_vk_core PUBLIC nss_frontend nss_vk_headers nss_volk nss_vk_shaders)
set_target_properties(nss_vk_core PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
)
if(WIN32)
  target_compile_definitions(nss_vk_core PUBLIC NOMINMAX)
endif()

add_library(nss_vk SHARED plugin.cpp)
set_target_properties(nss_vk PROPERTIES
  PREFIX "lib"
  OUTPUT_NAME "nss_vk"
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
)
target_compile_definitions(nss_vk PRIVATE "NSS_VERSION_STRING=\"${NSS_GIT_DESCRIBE}\"")
target_link_libraries(nss_vk PRIVATE nss_vk_core VapourSynth::VapourSynth)
